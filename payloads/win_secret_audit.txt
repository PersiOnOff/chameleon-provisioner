# Windows Environment & History Secret Hygiene Audit
# Проверка переменных окружения и истории PowerShell на забытые ключи и токены
USB_ID 046D:C31C
MFR Logitech
PROD USB Keyboard
ALT_MODE 1
Write-Host '=== Checking Environment Variables for Secrets ===' -ForegroundColor Yellow
Get-ChildItem Env: | Where-Object { $_.Name -match 'TOKEN|KEY|SECRET|PASS|PWD|API' } | Format-Table Name -AutoSize
Write-Host '=== Checking Recent PowerShell History ===' -ForegroundColor Yellow
$hist = (Get-PSReadlineOption).HistorySavePath
if (Test-Path $hist) { Get-Content $hist -Tail 20 | Select-String -Pattern 'password|token|secret|key|curl|invoke' }
