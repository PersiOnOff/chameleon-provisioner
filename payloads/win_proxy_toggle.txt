# Настройка локального прокси и проверка шлюза на Windows
USB_ID 046D:C31C
MFR Logitech
PROD USB Keyboard
ALT_MODE 1
Set-ItemProperty -Path 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Internet Settings' -Name ProxyEnable -Value 1
Set-ItemProperty -Path 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Internet Settings' -Name ProxyServer -Value '127.0.0.1:10808'
Write-Host 'Windows System Proxy enabled (127.0.0.1:10808)' -ForegroundColor Cyan
Get-NetIPConfiguration | Select-Object InterfaceAlias, IPv4Address, DNSServer
